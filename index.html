async function initFirebase() { const app = initializeApp(firebaseConfig); db = getFirestore(app); auth = getAuth(app); onAuthStateChanged(auth, async (u) => { if (isHandlingConflict) return; if (mealSubscription) { mealSubscription(); mealSubscription = null; } if (weightSubscription) { weightSubscription(); weightSubscription = null; } let curr = u; if (!curr) { try { const result = await signInAnonymously(auth); curr = result.user; } catch(e){ console.error("Auto-login failed:", e); // 顯示錯誤但不崩潰，允許使用者稍後重試或檢查設定 showMessage(`初始化失敗: ${getFriendlyErrorMessage(e)}`, true); return; } } if (curr) { userId = curr.uid; updateAuthUI(curr); loadUserProfile(curr); mealSubscription = subscribeToMealsAndData(userId); weightSubscription = subscribeToWeight(userId); } }); }
