// --- Google Fit Sync Logic --- window.syncGoogleFitData = async () => { // 1. 如果沒有 Token，嘗試自動重新登入獲取 if (!googleAccessToken) { const proceed = confirm("為了讀取 Google Fit 體重數據，需要您再次確認 Google 授權。\n\n點擊「確定」後將彈出登入視窗，請選擇您的帳號。"); if (!proceed) return; await signInWithGoogle(); // 如果登入後還是沒有 Token (例如使用者取消)，則中止 if (!googleAccessToken) return showMessage('未取得授權，無法同步', true); } const btn = document.querySelector('button[onclick="syncGoogleFitData()"]'); const originalText = btn.innerHTML; btn.innerHTML = '讀取中...'; btn.disabled = true; try { // Fetch aggregate data for the last 30 days to find the latest weight const end = Date.now(); const start = end - (30 * 24 * 60 * 60 * 1000); const response = await fetch('https://www.googleapis.com/fitness/v1/users/me/dataset:aggregate', { method: 'POST', headers: { 'Authorization': `Bearer ${googleAccessToken}`, 'Content-Type': 'application/json' }, body: JSON.stringify({ aggregateBy: [{ dataTypeName: "com.google.weight" }], startTimeMillis: start, endTimeMillis: end }) }); if (!response.ok) { const errDetail = await response.json(); console.error("Google Fit Error:", errDetail); // 特別偵測 403 錯誤，通常是 API 未啟用 if (response.status === 403) { throw new Error('請至 Google Cloud Console 啟用 "Fitness API"'); } throw new Error(`API 錯誤 (${response.status})`); } const data = await response.json(); // Find the latest data point let latestWeight = null; let latestTime = 0; if (data.bucket && data.bucket.length > 0) { for (const bucket of data.bucket) { if (bucket.dataset && bucket.dataset.length > 0) { for (const ds of bucket.dataset) { if (ds.point && ds.point.length > 0) { for (const p of ds.point) { if (p.value && p.value.length > 0) { // com.google.weight value is in kg (fpVal) // 找時間最晚的一筆 const pTime = parseInt(p.startTimeNanos || p.endTimeNanos) / 1000000; if (pTime > latestTime) { latestWeight = p.value[0].fpVal; latestTime = pTime; } } } } } } } } if (latestWeight) { // Update input and save document.getElementById('newWeightInput').value = latestWeight.toFixed(1); await saveWeightLog(); // This handles saving to Firestore and updating UI showMessage(`已同步最新體重: ${latestWeight.toFixed(1)}kg`, false); } else { showMessage('Google Fit 最近 30 天無體重資料', true); } } catch (error) { console.error(error); showMessage(`同步失敗: ${error.message}`, true); } finally { btn.innerHTML = originalText; btn.disabled = false; } };
